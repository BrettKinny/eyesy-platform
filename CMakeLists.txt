cmake_minimum_required(VERSION 3.16)
project(eyesy_core LANGUAGES CXX)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(eyesy_core engine/src/core.cpp engine/src/input_workflow.cpp)
target_include_directories(eyesy_core PUBLIC engine/src)
target_compile_options(eyesy_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE eyesy_core)
add_test(NAME core COMMAND core_tests)
add_executable(wav_tests tests/wav_tests.cpp)
target_link_libraries(wav_tests PRIVATE eyesy_core)
add_test(NAME wav COMMAND wav_tests)
add_executable(dsp_regression tests/dsp_regression.cpp)
target_link_libraries(dsp_regression PRIVATE eyesy_core)
add_test(NAME dsp_regression COMMAND dsp_regression)
add_executable(audio_ring_stress tests/audio_ring_stress.cpp)
target_link_libraries(audio_ring_stress PRIVATE eyesy_core Threads::Threads)
add_test(NAME audio_ring_stress COMMAND audio_ring_stress)
set_tests_properties(audio_ring_stress PROPERTIES TIMEOUT 15)
add_executable(wav_mutation_stress tests/wav_mutation_stress.cpp)
target_link_libraries(wav_mutation_stress PRIVATE eyesy_core)
add_test(NAME wav_mutation_stress COMMAND wav_mutation_stress)
set_tests_properties(wav_mutation_stress PROPERTIES TIMEOUT 15)
